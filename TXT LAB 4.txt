\section*{Entregable 1: Creación de la base de datos}

\subsection*{Entorno de ejecución}

Las consultas de este laboratorio fueron ejecutadas en
\textbf{OneCompiler} (\url{https://onecompiler.com/mysql}),
una plataforma en línea que provee un entorno MySQL 8 sin necesidad
de instalar software localmente. El motor utilizado es MySQL, por
lo que toda la sintaxis (tipos de datos, \texttt{AUTO\_INCREMENT},
\texttt{ENUM}, \texttt{FOREIGN KEY}, etc.) corresponde a ese estándar
y no es compatible con SQLite u otros gestores.

\subsection*{Script SQL}

El siguiente script crea la base de datos \texttt{austral} desde cero.
Primero se definen las tablas respetando el orden de las claves
foráneas: \texttt{colecciones} y \texttt{autores} antes que
\texttt{libros}, y \texttt{libros} antes que \texttt{pagos}.
Luego se insertan los datos oficiales del enunciado.
Los resultados de cada consulta se encuentran en el Anexo A.

\begin{lstlisting}[style=sql]
CREATE TABLE colecciones (
    id_coleccion      INT AUTO_INCREMENT,
    titulo            VARCHAR(120) NOT NULL,
    fecha_publicacion DATE         NOT NULL,
    PRIMARY KEY (id_coleccion)
);

CREATE TABLE autores (
    id_autor INT AUTO_INCREMENT,
    nombre   VARCHAR(80)  NOT NULL,
    pais     VARCHAR(60),
    genero   ENUM('Comic','Ensayo','Infantil','Novela','Poesia') NOT NULL,
    PRIMARY KEY (id_autor)
);

CREATE TABLE libros (
    id_libro            INT AUTO_INCREMENT,
    id_autor            INT          NOT NULL,
    id_coleccion        INT          NOT NULL,
    titulo              VARCHAR(120) NOT NULL,
    paginas             INT,
    ejemplares_vendidos INT,
    anio                INT,
    PRIMARY KEY (id_libro),
    FOREIGN KEY (id_autor)     REFERENCES autores(id_autor),
    FOREIGN KEY (id_coleccion) REFERENCES colecciones(id_coleccion)
);

CREATE TABLE pagos (
    id_pago    INT AUTO_INCREMENT,
    id_autor   INT           NOT NULL,
    fecha_pago DATE          NOT NULL,
    monto      DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (id_pago),
    FOREIGN KEY (id_autor) REFERENCES autores(id_autor)
);

\section*{Entregable 2: Consultas SQL}

Los outputs de cada consulta se encuentran en el Anexo A
(páginas \pageref{sec:anexo} en adelante).

% -------
\subsection*{Consulta 1}

La comparación \texttt{pais <> 'Chile'} descarta automáticamente
los valores NULL, ya que MySQL no evalúa NULL con operadores de
comparación: cualquier operación contra NULL devuelve NULL, que
el \texttt{WHERE} trata como falso. Por eso se agrega
\texttt{OR pais IS NULL} para incluir a Kenji Arakawa.

\begin{lstlisting}[style=sql]
SELECT nombre, genero
FROM   autores
WHERE  pais <> 'Chile' OR pais IS NULL
ORDER BY nombre;
\end{lstlisting}

% -------
\subsection*{Consulta 2}

\begin{lstlisting}[style=sql]
SELECT titulo, paginas
FROM   libros
WHERE  paginas BETWEEN 190 AND 220
ORDER BY paginas;
\end{lstlisting}

% -------
\subsection*{Consulta 3}

\begin{lstlisting}[style=sql]
SELECT COUNT(*)                 AS total_libros,
       SUM(ejemplares_vendidos) AS total_ejemplares,
       AVG(paginas)             AS promedio_paginas,
       MIN(anio)                AS anio_mas_antiguo,
       MAX(ejemplares_vendidos) AS max_ejemplares
FROM   libros;
\end{lstlisting}

% -------
\subsection*{Consulta 4}

\begin{lstlisting}[style=sql]
SELECT DISTINCT genero
FROM   autores
ORDER BY genero;
\end{lstlisting}

% -------
\subsection*{Consulta 5}

\begin{lstlisting}[style=sql]
SELECT a.nombre, COUNT(*) AS libros_2024
FROM   autores a
JOIN   libros  l ON a.id_autor = l.id_autor
WHERE  l.anio = 2024
GROUP BY a.id_autor, a.nombre
HAVING COUNT(*) > 1
ORDER BY a.nombre;
\end{lstlisting}

% -------
\subsection*{Consulta 6}

\begin{lstlisting}[style=sql]
SELECT titulo, ejemplares_vendidos
FROM   libros
WHERE  anio = 2024 AND ejemplares_vendidos > 2000
ORDER BY ejemplares_vendidos DESC
LIMIT 3;
\end{lstlisting}

% -------
\subsection*{Consulta 7}

\begin{lstlisting}[style=sql]
-- INSERT del libro temporal
INSERT INTO libros (id_autor, id_coleccion, titulo,
                    paginas, ejemplares_vendidos, anio)
VALUES (3, 1, 'Papel Mojado', 260, 50, 2024);

-- SELECT previo: ver la fila antes de borrarla
SELECT * FROM libros WHERE titulo = 'Papel Mojado';

-- DELETE filtrado por titulo
DELETE FROM libros WHERE titulo = 'Papel Mojado';

-- SELECT posterior: verificar que ya no esta
SELECT * FROM libros WHERE titulo = 'Papel Mojado';
\end{lstlisting}

% -------
\subsection*{Consulta 8}

\begin{lstlisting}[style=sql]
-- SELECT previo
SELECT * FROM libros WHERE titulo = 'Cielo de Enero';

-- UPDATE: suma 100 al valor actual
UPDATE libros
SET    ejemplares_vendidos = ejemplares_vendidos + 100
WHERE  titulo = 'Cielo de Enero';

-- SELECT posterior
SELECT * FROM libros WHERE titulo = 'Cielo de Enero';
\end{lstlisting}

% -------
\subsection*{Consulta 9}

\begin{lstlisting}[style=sql]
SELECT l.titulo            AS libro,
       c.titulo            AS coleccion,
       c.fecha_publicacion
FROM   libros      l
JOIN   colecciones c ON l.id_coleccion = c.id_coleccion;
\end{lstlisting}

% -------
\subsection*{Consulta 10}

\begin{lstlisting}[style=sql]
-- 10a: lista de pagos con nombre del autor
SELECT a.nombre, p.fecha_pago, p.monto
FROM   pagos   p
JOIN   autores a ON p.id_autor = a.id_autor;

-- 10b: total pagado por autor
SELECT a.nombre, SUM(p.monto) AS total_pagado
FROM   pagos   p
JOIN   autores a ON p.id_autor = a.id_autor
GROUP BY a.id_autor, a.nombre;
\end{lstlisting}
